   324→   324→func _update_rivals(delta: float, player_spd: float) -> void:
   325→   325→        if rivals.is_empty():
   326→   326→                return
   327→   327→        var racing := not ended and intro <= 0.0
   328→   328→        var rank := 1
   329→   329→        for R in rivals:
   330→   330→                if racing and float(R["wx"]) > world_x:
   331→   331→                        rank += 1
   332→   332→        finish_rank = rank
   333→   333→        for R in rivals:
   334→   334→                var n: Node2D = R["node"]
   335→   335→                if racing:
   336→   336→                        # سرعت با نوسان زنده + کش‌وسان: عقب بماند جانی می‌گیرد، زیاد جلو بزند کم می‌آید
   337→   337→                        var v := player_spd * float(R["pace"]) * (1.0 + 0.10 * sin(elapsed * 1.7 + float(R["phase"])))
   338→   338→                        var gap := world_x - float(R["wx"])
   339→   339→                        if gap > 420.0:
   340→   340→                                v *= 1.16
   341→   341→                        elif gap < -380.0:
   342→   342→                                v *= 0.88
   343→   343→                        R["wx"] = float(R["wx"]) + v * delta
   344→   344→                var sx := clampf(CAR_X + (float(R["wx"]) - world_x), -150.0, VIEW_W - 30.0)
   345→   345→                var lane_i: int = R["lane"]
   346→   346→                var spr: Sprite2D = R["spr"]
   347→   347→                if racing and not bool(R["air"]):
   348→   348→                        R["hop_in"] = float(R["hop_in"]) - delta
   349→   349→                        # مغز کوچک: مخروط جلوی راهش آمده → خودش می‌پرد

   352→   352→                                R["vy"] = -randf_range(520.0, 780.0)

   363→   363→                if bool(R["air"]):

   372→   372→                var th := spr.texture.get_height() * spr.scale.y

   633→   633→func _process(delta: float) -> void:
   634→   634→        if ended or get_tree().paused:

   636→   636→        elapsed += delta
   637→   637→
   638→   638→        # شمارش معکوس شروع — دنیا نفس نگه می‌دارد، ماشین زنده است

   641→   641→                var stage := int((2.0 - maxf(intro, 0.0)) / 0.5)
   642→   642→                if stage != _intro_stage and stage <= 3:
   643→   643→                        _intro_stage = stage

   652→   652→                        if stage >= 3:
   653→   653→                                AudioMgr.play_sfx("horn") # بوق شروع — سبقت بگیر!
   654→   654→                                cam.zoom = Vector2(1.13, 1.13) # پانچ دوربین لحظه‌ی پرتاب
   655→   655→                if intro <= 0.0:
   656→   656→                        intro_label.visible = false
   657→   657→                # موتور موقع گرم‌کردن دورش بالا می‌رود — هیجان از همین‌جا شروع می‌شود
   658→   658→                AudioMgr.set_engine(true, clampf(0.9 - intro * 0.35, 0.1, 0.9))
   659→   659→                cam.zoom = cam.zoom.lerp(Vector2.ONE, delta * 2.0)
   660→   660→                return
   661→   661→
   662→   662→        if autotest:

   691→   691→                target_mult = 1.85
   692→   692→        if elapsed < slow_until:
   693→   693→                target_mult = 0.38

   695→   695→        var spd := speed_base * speed_mult
   696→   696→
   697→   697→        world_x += spd * delta
   698→   698→        time_left -= delta

   700→   700→                _finish(false)

   702→   702→
   703→   703→        # scroll painterly parallax layers (mirror-tiled)

   722→   722→                vy = 0.0

   726→   726→        car.position = Vector2(CAR_X, GROUND_Y + car_y)
   727→   727→        car_sprite.rotation = clamp(vy * 0.00026, -0.20, 0.12) # دماغ‌کوبی ممنوع — سقوط نرم

   732→   732→
   733→   733→        # لنگرِ حباب گفتار — بالای سر ماشین (مختصات صفحه)
