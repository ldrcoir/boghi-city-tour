   242→func _add_gradient(top: bool) -> void:

   252→                g.colors = PackedColorArray([Color(0, 0, 0, 0.0), Color(0.04, 0.03, 0.05, 0.72)])
   253→                gt.fill_from = Vector2(0, 0)
   254→                gt.fill_to = Vector2(0, 1)

   257→        tr.texture = gt

   616→func _layout_watchdog() -> void:
   617→        # نگهبان چیدمان: اگر پنل منو روی دید نیست، هندسه‌ی واقعی را روی صفحه نشان بده
   618→        # تا کاربر عکس بگیرد و ما مقادیر واقعی گوشی را ببینیم
   619→        await get_tree().create_timer(1.6).timeout
   620→        if not is_instance_valid(panel_home):
   621→                return
   622→        # اگر کاربر به پنل دیگری رفته، نظارت فقط مال صفحه‌ی خانه است
   623→        if not panel_home.visible:
   624→                return
   625→        var vp := get_viewport_rect().size
   626→        var gr: Rect2 = panel_home.get_global_rect()
   627→        var on_screen: bool = panel_home.visible and gr.size.x > 60.0 and gr.size.y > 60.0 \
   628→                and gr.position.x < vp.x - 60.0 and gr.position.y < vp.y - 60.0 \
   629→                and gr.end.x > 60.0 and gr.end.y > 60.0
   630→        if on_screen:
   631→                print("[boghi][watchdog] panel OK rect=", gr, " viewport=", vp)
   632→                return
   633→        var msg := "خطای چیدمان — لطفاً از این صفحه عکس بگیر و بفرست\n"
   634→        msg += "viewport=%s window=%s\n" % [vp, get_window().size]
   635→        msg += "root=%s\npanel_global=%s visible=%s\n" % [Rect2(Vector2.ZERO, size), gr, panel_home.visible]
   636→        msg += "locale=%s scale=%s\n" % [OS.get_locale(), get_viewport().content_scale_factor]
   637→        print("[boghi][watchdog] ", msg.replace("\n", " | "))
   638→        var box := PanelContainer.new()
   639→        var sbf := StyleBoxFlat.new()
   640→        sbf.bg_color = Color(0.25, 0.05, 0.05, 0.94)
   641→        sbf.set_corner_radius_all(12)
   642→        box.add_theme_stylebox_override("panel", sbf)
   643→        box.set_anchors_preset(Control.PRESET_FULL_RECT)
   644→        box.mouse_filter = Control.MOUSE_FILTER_IGNORE
   645→        var lbl := _mk_label(msg, 22, true, false, Color(1, 0.9, 0.85))
   646→        lbl.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
   647→        box.add_child(lbl)
   648→        add_child(box)

   650→func _show(p: PanelContainer) -> void:
   651→        if panel_home == null or panel_levels == null:
   652→                return
   653→        panel_home.visible = p == panel_home
   654→        panel_levels.visible = p == panel_levels
   655→        # هر باز شدن مأموریت‌ها: تیکه تازه‌ی اکبر + صدای لاله‌زبانش
   656→        if p == panel_levels and akbar_line_label != null:
   657→                var line := CarBrain.pick_line("akbar", "idle")
   658→                if line.is_empty():
   659→                        line = "امروز جاده مال ماست!"
   660→                akbar_line_label.text = line
   661→                if brain != null and akbar_rect != null and is_instance_valid(akbar_rect) and akbar_rect.texture != null:
   662→                        brain.say(akbar_rect, "akbar", "idle")
   663→        # کانتینر مأموریت‌ها هم وقتی پنل مخفی است باید مخفی شود
   664→        # (دو لایه محافظت: mouse_filter=IGNORE + مخفی‌سازی کامل)
   665→        # ماشین‌های تزئینی و دکمه‌های تپ فقط مال صفحه‌ی خانه‌اند؛
   666→        # وگرنه روی پنل مأموریت‌ها می‌افتند و تپِ دکمه‌ها را می‌دزدند
   667→        for n in deco_cars:
   668→                if is_instance_valid(n):
   669→                        n.visible = p == panel_home
   670→        if levels_center != null:
   671→                levels_center.visible = p == panel_levels
   672→        _refresh()
