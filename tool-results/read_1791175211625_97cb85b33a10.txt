   639→func _show(p: PanelContainer) -> void:
   640→        if panel_home == null or panel_levels == null:
   641→                return
   642→        panel_home.visible = p == panel_home
   643→        panel_levels.visible = p == panel_levels
   644→        # تیتر فقط مال صفحه‌ی خانه است (لیست ماموریت‌ها خودش هدر دارد)
   645→        for n in title_nodes:
   646→                if is_instance_valid(n):
   647→                        n.visible = p == panel_home
   648→        for n in deco_cars:
   649→                if is_instance_valid(n):
   650→                        n.visible = p == panel_home
   651→        # هر باز شدن مأموریت‌ها: تیکه تازه‌ی اکبر
   652→        if p == panel_levels and akbar_line_label != null:
   653→                var line := CarBrain.pick_line("akbar", "idle")
   654→                if line.is_empty():
   655→                        line = "امروز جاده مال ماست!"
   656→                akbar_line_label.text = line
   657→                if brain != null and akbar_rect != null and is_instance_valid(akbar_rect) and akbar_rect.texture != null:
   658→                        brain.say(akbar_rect, "akbar", "idle")
   659→        _refresh()
