   365→func _unhandled_input(e: InputEvent) -> void:
   366→        if e is InputEventKey and e.pressed:
   367→                if e.keycode == KEY_SPACE or e.keycode == KEY_UP or e.keycode == KEY_W:
   368→                        do_jump()
   369→                elif e.keycode == KEY_X or e.keycode == KEY_SHIFT:
   370→                        _on_boost()

   372→func do_jump() -> void:
   373→        if ended:
   374→                return
   375→        if on_ground:
   376→                vy = -980.0 * float(Globals.car_stats()["jump"])
   377→                on_ground = false
   378→                AudioMgr.play_sfx("jump")
   379→
   380→func _on_boost() -> void:
   381→        if ended or turbo_meter < 0.99:
   382→                return
   383→        turbo_left = float(Globals.car_stats()["turbo"])
   384→        turbo_meter = 0.0
   385→        AudioMgr.play_sfx("boost")
   386→        if brain != null:
   387→                brain.say(car_anchor, str(Globals.CARS[Globals.selected_car]["id"]), "nitro")

   389→func _toggle_pause() -> void:
   390→        var t := get_tree()
   391→        t.paused = not t.paused
   392→        if end_panel == null:
   393→                pass

   395→func _process(delta: float) -> void:
   396→        if ended or get_tree().paused:
   397→                return
   398→        elapsed += delta
   399→        if autotest:
   400→                _auto_timer += delta
   401→                if _auto_timer > 1.1:
   402→                        _auto_timer = 0.0
   403→                        do_jump()
   404→                if elapsed > 3.2 and not _shot_taken:
   405→                        _shot_taken = true
   406→                        var img := get_viewport().get_texture().get_image()
   407→                        img.save_png("/home/z/my-project/scripts/shot_game.png")
   408→                if elapsed > 6.0:
   409→                        print("AUTOTEST OK world_x=", int(world_x), " coins=", coins_got, " objs=", world.get_child_count())
   410→                        get_tree().quit()
   411→        var boosting := turbo_left > 0.0
   412→        if boosting:
   413→                turbo_left -= delta
   414→        turbo_meter = min(1.0, turbo_meter + delta * 0.22)
   415→        var target_mult := 1.0
   416→        if boosting:
   417→                target_mult = 1.85
   418→        if elapsed < slow_until:
   419→                target_mult = 0.38
   420→        speed_mult = lerp(speed_mult, target_mult, delta * 4.0)
   421→        var spd := speed_base * speed_mult
   422→
   423→        world_x += spd * delta
   424→        time_left -= delta
   425→        if time_left <= 0.0:
   426→                _finish(false)
   427→                return
   428→
   429→        # scroll painterly parallax layers (mirror-tiled)

   440→        # car physics
   441→        vy += GRAVITY * delta
   442→        car_y += vy * delta

   445→                vy = 0.0

   514→func _spawn(x: float) -> void:
   515→        var rel := x - world_x
   516→        var coin_rate := float(lv["coin_rate"])
   517→        var obst_rate := float(lv["obstacle_rate"])
   518→        if lv["type"] == "collect":
   519→                obst_rate *= 0.5
   520→                coin_rate = min(1.0, coin_rate + 0.2)
   521→        # scheduled passengers for taxi
   522→        if next_passenger_idx < passenger_spots.size() and x >= passenger_spots[next_passenger_idx]:
   523→                next_passenger_idx += 1
   524→                _make_obj("passenger", rel)
   525→                return
   526→        var r := randf()
   527→        if r < obst_rate:
   528→                _make_obj("cone", rel)
   529→        elif r < obst_rate + coin_rate:
   530→                _make_coin_arc(rel)
   531→        elif r < obst_rate + coin_rate + float(lv["ramp_rate"]):
   532→                _make_obj("ramp", rel)

   780→func _make_shadow_tex() -> ImageTexture:
   781→    var sz := Vector2i(180, 56)
   782→    var img := Image.create(sz.x, sz.y, false, Image.FORMAT_RGBA8)
   783→    var cx := sz.x / 2.0
   784→    var cy := sz.y / 2.0
   785→    for y in sz.y:
   786→        for x in sz.x:
   787→            var d := Vector2((x - cx) / (cx - 6.0), (y - cy) / (cy - 4.0)).length()
   788→            var a: float = clamp(1.0 - d, 0.0, 1.0)
   789→            img.set_pixel(x, y, Color(0.03, 0.02, 0.05, a * a * 0.9))
   790→    return ImageTexture.create_from_image(img)
