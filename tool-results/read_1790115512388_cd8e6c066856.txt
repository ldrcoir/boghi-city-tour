   347→func _unhandled_input(e: InputEvent) -> void:
   348→        if e is InputEventKey and e.pressed:
   349→                if e.keycode == KEY_SPACE or e.keycode == KEY_UP or e.keycode == KEY_W:
   350→                        do_jump()
   351→                elif e.keycode == KEY_X or e.keycode == KEY_SHIFT:
   352→                        _on_boost()

   354→func do_jump() -> void:
   355→        if ended:
   356→                return
   357→        if on_ground:
   358→                vy = -980.0 * float(Globals.car_stats()["jump"])
   359→                on_ground = false
   360→                AudioMgr.play_sfx("jump")

   362→func _on_boost() -> void:
   363→        if ended or turbo_meter < 0.99:
   364→                return
   365→        turbo_left = float(Globals.car_stats()["turbo"])
   366→        turbo_meter = 0.0
   367→        AudioMgr.play_sfx("boost")
   368→        if brain != null:
   369→                brain.say(car_anchor, str(Globals.CARS[Globals.selected_car]["id"]), "nitro")

   371→func _toggle_pause() -> void:
   372→        var t := get_tree()
   373→        t.paused = not t.paused
   374→        if end_panel == null:
   375→                pass

   377→func _process(delta: float) -> void:
   378→        if ended or get_tree().paused:
   379→                return
   380→        elapsed += delta
   381→        if autotest:
   382→                _auto_timer += delta
   383→                if _auto_timer > 1.1:
   384→                        _auto_timer = 0.0
   385→                        do_jump()
   386→                if elapsed > 3.2 and not _shot_taken:
   387→                        _shot_taken = true
   388→                        var img := get_viewport().get_texture().get_image()
   389→                        img.save_png("/home/z/my-project/scripts/shot_game.png")
   390→                if elapsed > 6.0:
   391→                        print("AUTOTEST OK world_x=", int(world_x), " coins=", coins_got, " objs=", world.get_child_count())
   392→                        get_tree().quit()
   393→        var boosting := turbo_left > 0.0
   394→        if boosting:
   395→                turbo_left -= delta
   396→        turbo_meter = min(1.0, turbo_meter + delta * 0.22)
   397→        var target_mult := 1.0
   398→        if boosting:
   399→                target_mult = 1.85
   400→        if elapsed < slow_until:
   401→                target_mult = 0.38
   402→        speed_mult = lerp(speed_mult, target_mult, delta * 4.0)
   403→        var spd := speed_base * speed_mult
   404→
   405→        world_x += spd * delta
   406→        time_left -= delta
   407→        if time_left <= 0.0:
   408→                _finish(false)
   409→                return
   410→
   411→        # scroll painterly parallax layers (mirror-tiled)
   412→        for L in layers:
   413→                var off := fmod(world_x * float(L["f"]), float(L["w"]))
   414→                (L["s1"] as Sprite2D).position.x = -off
   415→                (L["s2"] as Sprite2D).position.x = float(L["w"]) - off
   416→
   417→        # spawn
   418→        while world_x + VIEW_W > next_obj_x:
   419→                _spawn(next_obj_x)
   420→                next_obj_x += randf_range(260.0, 460.0) * (1.0 + float(lv["speed"]) / 70.0)
   421→
   422→        # car physics
   423→        vy += GRAVITY * delta
   424→        car_y += vy * delta
   425→        if car_y >= 0.0:
   426→                car_y = 0.0
   427→                vy = 0.0
   428→                if not on_ground:
   429→                        on_ground = true
   430→        car.position = Vector2(CAR_X, GROUND_Y + car_y)
   431→        car_sprite.rotation = clamp(vy * 0.00045, -0.3, 0.35)
   432→        var h: float = clamp(-car_y / 420.0, 0.0, 1.0)
   433→        shadow.position = Vector2(CAR_X, GROUND_Y - 6)
   434→        shadow.scale = Vector2(1.0 - 0.4 * h, 1.0 - 0.25 * h)
   435→        shadow.modulate = Color(1, 1, 1, 0.5 - 0.32 * h)
   436→
   437→        # لنگرِ حباب گفتار — بالای سر ماشین (مختصات صفحه)
   438→        if car_anchor != null:
   439→                var sp: Vector2 = car_sprite.get_global_transform_with_canvas().origin
   440→                car_anchor.position = sp + Vector2(0.0, -140.0)
   441→
   442→        # بوست مخفی بوقی — سیستم طرف بوقی است! هر چند ثانیه نیترویش را پر می‌کند
   443→        if brain != null and Globals.selected_car == 0 and not autotest:

   496→func _spawn(x: float) -> void:
   497→        var rel := x - world_x
   498→        var coin_rate := float(lv["coin_rate"])
   499→        var obst_rate := float(lv["obstacle_rate"])
   500→        if lv["type"] == "collect":
   501→                obst_rate *= 0.5
   502→                coin_rate = min(1.0, coin_rate + 0.2)
   503→        # scheduled passengers for taxi
   504→        if next_passenger_idx < passenger_spots.size() and x >= passenger_spots[next_passenger_idx]:
   505→                next_passenger_idx += 1
   506→                _make_obj("passenger", rel)
   507→                return
   508→        var r := randf()
   509→        if r < obst_rate:
   510→                _make_obj("cone", rel)
   511→        elif r < obst_rate + coin_rate:
   512→                _make_coin_arc(rel)
   513→        elif r < obst_rate + coin_rate + float(lv["ramp_rate"]):
   514→                _make_obj("ramp", rel)

   673→func _make_shadow_tex() -> ImageTexture:
   674→    var sz := Vector2i(180, 56)
   675→    var img := Image.create(sz.x, sz.y, false, Image.FORMAT_RGBA8)
   676→    var cx := sz.x / 2.0
   677→    var cy := sz.y / 2.0
   678→    for y in sz.y:
   679→        for x in sz.x:
   680→            var d := Vector2((x - cx) / (cx - 6.0), (y - cy) / (cy - 4.0)).length()
   681→            var a: float = clamp(1.0 - d, 0.0, 1.0)
   682→            img.set_pixel(x, y, Color(0.03, 0.02, 0.05, a * a * 0.9))
   683→    return ImageTexture.create_from_image(img)
