   291→func _spawn_rivals() -> void:
   292→        var pool := RIVAL_POOL.duplicate()

   298→                        break
   299→                var tex_path := "res://assets/sprites/" + str(pool[i]) + "_side.png"

   301→                        continue

   304→                var spr := Sprite2D.new()
   305→                spr.texture = load(tex_path)
   306→                var sc := (214.0 * (0.84 if lane == 0 else 1.0)) / float(spr.texture.get_height())
   307→                spr.scale = Vector2(sc, sc)
   308→                var sh := Sprite2D.new()
   309→                sh.texture = _make_shadow_tex()
   310→                sh.position = Vector2(0, -6)
   311→                sh.z_index = -2

   315→                add_child(n)
   316→                rivals.append({
   317→                        "node": n, "spr": spr, "sh": sh,
   318→                        "wx": float(start_wx[i]), "ry": 0.0, "vy": 0.0, "air": false,
   319→                        "lane": lane, "phase": randf() * TAU,
   320→                        "pace": randf_range(0.94, 1.06),

   324→func _update_rivals(delta: float, player_spd: float) -> void:
   325→        if rivals.is_empty():
   326→                return
   327→        var racing := not ended and intro <= 0.0
   328→        var rank := 1
   329→        for R in rivals:
   330→                if racing and float(R["wx"]) > world_x:
   331→                        rank += 1
   332→        finish_rank = rank
   333→        for R in rivals:
   334→                var n: Node2D = R["node"]
   335→                if racing:
   336→                        # سرعت با نوسان زنده + کش‌وسان: عقب بماند جانی می‌گیرد، زیاد جلو بزند کم می‌آید
   337→                        var v := player_spd * float(R["pace"]) * (1.0 + 0.10 * sin(elapsed * 1.7 + float(R["phase"])))
   338→                        var gap := world_x - float(R["wx"])
   339→                        if gap > 420.0:
   340→                                v *= 1.16
   341→                        elif gap < -380.0:
   342→                                v *= 0.88
   343→                        R["wx"] = float(R["wx"]) + v * delta
   344→                var sx := clampf(CAR_X + (float(R["wx"]) - world_x), -150.0, VIEW_W - 30.0)
   345→                var lane_i: int = R["lane"]
   346→                var spr: Sprite2D = R["spr"]
   347→                if racing and not bool(R["air"]):
   348→                        R["hop_in"] = float(R["hop_in"]) - delta
   349→                        # مغز کوچک: مخروط جلوی راهش آمده → خودش می‌پرد
   350→                        if float(R["hop_in"]) <= 0.0:
   351→                                R["hop_in"] = randf_range(5.0, 11.0)
   352→                                R["vy"] = -randf_range(520.0, 780.0)
   353→                                R["air"] = true
   354→                        else:
   355→                                for obj in world.get_children():
   356→                                        if str(obj.get_meta("kind", "")) == "cone":
   357→                                                var ox := float(obj.position.x)
   358→                                                if ox > sx - 30.0 and ox < sx + 200.0:
   359→                                                        R["vy"] = -randf_range(640.0, 820.0)
   360→                                                        R["air"] = true
   361→                                                        break
   362→                var ry: float = R["ry"]
   363→                if bool(R["air"]):
   364→                        R["vy"] = float(R["vy"]) + GRAVITY * delta
   365→                        ry += float(R["vy"]) * delta
   366→                        if ry >= 0.0:
   367→                                ry = 0.0
   368→                                R["air"] = false
   369→                R["ry"] = ry
   370→                var base_y := GROUND_Y - 24.0 if lane_i == 0 else GROUND_Y + 8.0
   371→                n.position = Vector2(sx, base_y)
   372→                var th := spr.texture.get_height() * spr.scale.y
   373→                spr.position.y = 12.0 - th * 0.5 - ry + sin(elapsed * 21.0 + float(R["phase"])) * 2.2
   374→                spr.rotation = clamp(float(R["vy"]) * 0.00022, -0.15, 0.10)
   375→                var hh: float = clamp(-ry / 420.0, 0.0, 1.0)
   376→                var sh2: Sprite2D = R["sh"]
   377→                sh2.position = Vector2(0, -6)
   378→                sh2.scale = Vector2(spr.scale.x / spr.scale.y * (1.0 - 0.4 * hh), 1.0 - 0.25 * hh)
   379→                sh2.modulate = Color(1, 1, 1, 0.5 - 0.32 * hh)
   380→        if pos_label != null:
   381→                pos_label.text = Globals.L("pos") + " " + str(rank) + "/" + str(rivals.size() + 1)

   383→## اسکواش و استرچ — ماشین جان دارد: پرتاب کش می‌آید، فرود له می‌شود
   384→func _car_squash(sx: float, sy: float) -> void:
   385→        if car_sprite == null:
   386→                return
   387→        if _sq_tw != null and _sq_tw.is_valid():
   388→                _sq_tw.kill()
   389→        car_sprite.scale = car_scale_base * Vector2(sx, sy)
   390→        _sq_tw = create_tween()
   391→        _sq_tw.tween_property(car_sprite, "scale", car_scale_base, 0.28).set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)

   633→func _process(delta: float) -> void:
   634→        if ended or get_tree().paused:
   635→                return
   636→        elapsed += delta
   637→
   638→        # شمارش معکوس شروع — دنیا نفس نگه می‌دارد، ماشین زنده است
   639→        if intro > 0.0:
   640→                intro -= delta
   641→                var stage := int((2.0 - maxf(intro, 0.0)) / 0.5)
   642→                if stage != _intro_stage and stage <= 3:
   643→                        _intro_stage = stage
   644→                        intro_label.text = str(3 - stage) if stage < 3 else Globals.L("go")
   645→                        # مرکز چرخش مستقل از layout — گوشه‌ی صحنه هرگز
   646→                        intro_label.pivot_offset = Vector2(get_viewport_rect().size.x * 0.5, 85.0)
   647→                        intro_label.scale = Vector2(1.7, 1.7)
   648→                        intro_label.modulate.a = 1.0
   649→                        var itw := create_tween()
   650→                        itw.tween_property(intro_label, "scale", Vector2.ONE, 0.3).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
   651→                        itw.parallel().tween_property(intro_label, "modulate:a", 0.0, 0.45).set_delay(0.2)
   652→                        if stage >= 3:
   653→                                AudioMgr.play_sfx("horn") # بوق شروع — سبقت بگیر!
   654→                                cam.zoom = Vector2(1.13, 1.13) # پانچ دوربین لحظه‌ی پرتاب
   655→                if intro <= 0.0:
   656→                        intro_label.visible = false
   657→                # موتور موقع گرم‌کردن دورش بالا می‌رود — هیجان از همین‌جا شروع می‌شود
   658→                AudioMgr.set_engine(true, clampf(0.9 - intro * 0.35, 0.1, 0.9))
   659→                cam.zoom = cam.zoom.lerp(Vector2.ONE, delta * 2.0)
   660→                return
   661→
   662→        if autotest:
   663→                _auto_timer += delta
   664→                # تپ واقعی روی نیمه‌صفحه‌ی راست — باید ماشین را از زمین بلند کند
   665→                if _auto_timer > 0.8 and not _zone_tested:
   666→                        _zone_tested = true
   667→                        _tap_at(Vector2(VIEW_W * 0.78, VIEW_H * 0.55))
   668→                if _auto_timer > 1.15 and not _zone_checked:
   669→                        _zone_checked = true
   670→                        var air := not on_ground
   671→                        print("[boghi][autotest] TAP-ZONE-JUMP ", "OK" if air else "FAIL")
   672→                        if not air:
   673→                                get_tree().quit(1)
   674→                                return
   675→                if _auto_timer > 2.3:
   676→                        _auto_timer = 0.0
   677→                        do_jump()
   678→                if elapsed > 3.4 and not _shot_taken:
   679→                        _shot_taken = true
   680→                        var img := get_viewport().get_texture().get_image()
   681→                        img.save_png("/home/z/my-project/scripts/shot_game.png")
   682→                if elapsed > 6.0:
   683→                        print("AUTOTEST OK world_x=", int(world_x), " coins=", coins_got, " objs=", world.get_child_count(), " rivals=", rivals.size(), " rank=", finish_rank)
   684→                        get_tree().quit()
   685→        var boosting := turbo_left > 0.0
   686→        if boosting:
   687→                turbo_left -= delta
   688→        turbo_meter = min(1.0, turbo_meter + delta * 0.22)
   689→        var target_mult := 1.0
   690→        if boosting:
   691→                target_mult = 1.85
   692→        if elapsed < slow_until:
   693→                target_mult = 0.38
   694→        speed_mult = lerp(speed_mult, target_mult, delta * 4.0)
   695→        var spd := speed_base * speed_mult
   696→
   697→        world_x += spd * delta
   698→        time_left -= delta
   699→        if time_left <= 0.0:
   700→                _finish(false)
   701→                return
   702→
   703→        # scroll painterly parallax layers (mirror-tiled)
   704→        for L in layers:
   705→                var off := fmod(world_x * float(L["f"]), float(L["w"]))
   706→                (L["s1"] as Sprite2D).position.x = -off
   707→                (L["s2"] as Sprite2D).position.x = float(L["w"]) - off
   708→
   709→        # رقیب‌ها نفس می‌کشند — مسابقه یعنی جدال، نه رانندگی تنها

   716→
   717→        # car physics
   718→        vy += GRAVITY * delta
   719→        car_y += vy * delta
   720→        if car_y >= 0.0:
   721→                car_y = 0.0
   722→                vy = 0.0
   723→                if not on_ground:
   724→                        on_ground = true
   725→                        _on_land() # له‌شدن فرود + گردوخاک + لرزش ریز
   726→        car.position = Vector2(CAR_X, GROUND_Y + car_y)
   727→        car_sprite.rotation = clamp(vy * 0.00026, -0.20, 0.12) # دماغ‌کوبی ممنوع — سقوط نرم
   728→        var h: float = clamp(-car_y / 420.0, 0.0, 1.0)

   730→        shadow.scale = Vector2(1.0 - 0.4 * h, 1.0 - 0.25 * h)

   732→
   733→        # لنگرِ حباب گفتار — بالای سر ماشین (مختصات صفحه)
   734→        if car_anchor != null:
   735→                var sp: Vector2 = car_sprite.get_global_transform_with_canvas().origin
   736→                car_anchor.position = sp + Vector2(0.0, -140.0)
   737→
   738→        # بوست مخفی بوقی — سیستم طرف بوقی است! هر چند ثانیه نیترویش را پر می‌کند
   739→        if brain != null and Globals.selected_car == 0 and not autotest:
   740→                _boost_in -= delta
   741→                if _boost_in <= 0.0 and turbo_meter < 0.62:
   742→                        _boost_in = 13.0 + randf() * 7.0
   743→                        turbo_meter = 1.0
   744→                        nitro_bar.modulate = Color(1.6, 1.4, 0.7)
   745→                        var tw := create_tween()
   746→                        tw.tween_property(nitro_bar, "modulate", Color(1, 1, 1), 0.8)
   747→                        brain.say(car_anchor, "boghi", "boost")
   748→
   749→        # move world objects
   750→        for obj in world.get_children():
   751→                obj.position.x -= spd * delta
   752→                if obj.position.x < -260:
   753→                        obj.queue_free()
   754→
   755→        # hud
   756→        coin_label.text = "سکه " + str(coins_got) + (" / " + str(target_coins) if target_coins > 0 else "")
   757→        time_label.text = "زمان " + str(int(ceil(time_left)))
   758→        progress.value = 100.0 * world_x / finish_px
   759→        if target_passengers > 0:
   760→                extra_label.text = "مسافر " + str(passengers) + " / " + str(target_passengers)
   761→        # جلوه‌های نیترو و زندگی صحنه
   762→        flame.emitting = boosting
   763→        dust.emitting = on_ground and spd > 70.0
   764→        lines.visible = boosting
   765→        nitro_bar.value = turbo_meter * 100.0
   766→        var zoom_target := Vector2(0.92, 0.92) if boosting else Vector2.ONE
   767→        cam.zoom = cam.zoom.lerp(zoom_target, delta * 3.0)
   768→        if on_ground:
   769→                car_sprite.position.y = 12.0 - car_sprite.texture.get_height() * car_sprite.scale.y * 0.5 + sin(elapsed * 26.0) * 2.4
   770→
   771→        boost_btn.modulate = Color(1, 1, 1) if turbo_meter >= 0.99 else Color(0.6, 0.6, 0.6)
   772→        AudioMgr.set_engine(true, clamp((spd / 300.0) * speed_mult, 0.0, 1.0))
   773→
   774→        # win conditions
   775→        var win := false
   776→        if lv["type"] == "collect" and target_coins > 0 and coins_got >= target_coins:
   777→                win = true
   778→        if lv["type"] == "taxi" and target_passengers > 0 and passengers >= target_passengers:
   779→                win = true
   780→        if world_x >= finish_px:
   781→                win = true
   782→        if win:
   783→                _finish(true)
   784→
   785→        # shake decay
   786→        if shake > 0.0:
   787→                shake = max(0.0, shake - delta * 30.0)
   788→                cam.offset = Vector2(randf_range(-shake, shake), randf_range(-shake, shake))
   789→        else:
   790→                cam.offset = Vector2.ZERO

   792→func _spawn(x: float) -> void:
   793→        var rel := x - world_x
   794→        var coin_rate := float(lv["coin_rate"])
   795→        var obst_rate := float(lv["obstacle_rate"])
   796→        if lv["type"] == "collect":
   797→                obst_rate *= 0.5
   798→                coin_rate = min(1.0, coin_rate + 0.2)
   799→        # scheduled passengers for taxi
   800→        if next_passenger_idx < passenger_spots.size() and x >= passenger_spots[next_passenger_idx]:
   801→                next_passenger_idx += 1
   802→                _make_obj("passenger", rel)
   803→                return
   804→        var r := randf()
   805→        if r < obst_rate:
   806→                _make_obj("cone", rel)
   807→        elif r < obst_rate + coin_rate:
   808→                _make_coin_arc(rel)
   809→        elif r < obst_rate + coin_rate + float(lv["ramp_rate"]):
   810→                _make_obj("ramp", rel)

  1113→func _make_shadow_tex() -> ImageTexture:
  1114→    var sz := Vector2i(180, 56)
  1115→    var img := Image.create(sz.x, sz.y, false, Image.FORMAT_RGBA8)
  1116→    var cx := sz.x / 2.0
  1117→    var cy := sz.y / 2.0
  1118→    for y in sz.y:
  1119→        for x in sz.x:
  1120→            var d := Vector2((x - cx) / (cx - 6.0), (y - cy) / (cy - 4.0)).length()
  1121→            var a: float = clamp(1.0 - d, 0.0, 1.0)
  1122→            img.set_pixel(x, y, Color(0.03, 0.02, 0.05, a * a * 0.9))
  1123→    return ImageTexture.create_from_image(img)
