   648→func _show(p: PanelContainer) -> void:
   649→        if panel_home == null or panel_levels == null:
   650→                return
   651→        panel_home.visible = p == panel_home
   652→        panel_levels.visible = p == panel_levels
   653→        # تیتر فقط مال صفحه‌ی خانه است (لیست ماموریت‌ها خودش هدر دارد)
   654→        for n in title_nodes:
   655→                if is_instance_valid(n):
   656→                        n.visible = p == panel_home
   657→        for n in deco_cars:
   658→                if is_instance_valid(n):
   659→                        n.visible = p == panel_home
   660→        # هر باز شدن مأموریت‌ها: تیکه تازه‌ی اکبر
   661→        if p == panel_levels and akbar_line_label != null:
   662→                var line := CarBrain.pick_line("akbar", "idle")
   663→                if line.is_empty():
   664→                        line = "امروز جاده مال ماست!"
   665→                akbar_line_label.text = line
   666→                if brain != null and akbar_rect != null and is_instance_valid(akbar_rect) and akbar_rect.texture != null:
   667→                        brain.say(akbar_rect, "akbar", "idle")
   668→        _refresh()
