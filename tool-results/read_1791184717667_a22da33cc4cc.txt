   141→   141→func _add_layer(path: String, z: int, bottom_y: float, scale: float, factor: float, squash: float = 0.0) -> void:
   142→   142→        var tex: Texture2D = load(path)
   143→   143→        var sc := Vector2(scale, scale if squash <= 0.0 else squash)
   144→   144→        var s1 := Sprite2D.new()
   145→   145→        s1.texture = tex
   146→   146→        s1.centered = false
   147→   147→        s1.z_index = z
   148→   148→        s1.scale = sc
   149→   149→        s1.position = Vector2(0.0, bottom_y - tex.get_height() * sc.y)
   150→   150→        add_child(s1)
   151→   151→        var s2 := s1.duplicate() as Sprite2D
   152→   152→        s2.flip_h = true
   153→   153→        s2.position = s1.position + Vector2(tex.get_width() * scale, 0.0)
   154→   154→        add_child(s2)
   155→   155→        layers.append({"s1": s1, "s2": s2, "w": tex.get_width() * scale, "f": factor})

   867→   867→func _make_shadow_tex() -> ImageTexture:
   868→   868→    var sz := Vector2i(180, 56)
   869→   869→    var img := Image.create(sz.x, sz.y, false, Image.FORMAT_RGBA8)
   870→   870→    var cx := sz.x / 2.0
   871→   871→    var cy := sz.y / 2.0
   872→   872→    for y in sz.y:
   873→   873→        for x in sz.x:
   874→   874→            var d := Vector2((x - cx) / (cx - 6.0), (y - cy) / (cy - 4.0)).length()
   875→   875→            var a: float = clamp(1.0 - d, 0.0, 1.0)
   876→   876→            img.set_pixel(x, y, Color(0.03, 0.02, 0.05, a * a * 0.9))
   877→   877→    return ImageTexture.create_from_image(img)
