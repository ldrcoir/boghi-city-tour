   240→func _add_gradient(top: bool) -> void:

   255→        tr.texture = gt

   585→func _layout_watchdog() -> void:
   586→        # نگهبان چیدمان: اگر پنل منو روی دید نیست، هندسه‌ی واقعی را روی صفحه نشان بده
   587→        # تا کاربر عکس بگیرد و ما مقادیر واقعی گوشی را ببینیم
   588→        await get_tree().create_timer(1.6).timeout
   589→        if not is_instance_valid(panel_home):
   590→                return
   591→        # اگر کاربر به پنل دیگری رفته، نظارت فقط مال صفحه‌ی خانه است
   592→        if not panel_home.visible:
   593→                return
   594→        var vp := get_viewport_rect().size
   595→        var gr: Rect2 = panel_home.get_global_rect()
   596→        var on_screen: bool = panel_home.visible and gr.size.x > 60.0 and gr.size.y > 60.0 \
   597→                and gr.position.x < vp.x - 60.0 and gr.position.y < vp.y - 60.0 \
   598→                and gr.end.x > 60.0 and gr.end.y > 60.0
   599→        if on_screen:
   600→                print("[boghi][watchdog] panel OK rect=", gr, " viewport=", vp)
   601→                return
   602→        var msg := "خطای چیدمان — لطفاً از این صفحه عکس بگیر و بفرست\n"
   603→        msg += "viewport=%s window=%s\n" % [vp, get_window().size]
   604→        msg += "root=%s\npanel_global=%s visible=%s\n" % [Rect2(Vector2.ZERO, size), gr, panel_home.visible]
   605→        msg += "locale=%s scale=%s\n" % [OS.get_locale(), get_viewport().content_scale_factor]
   606→        print("[boghi][watchdog] ", msg.replace("\n", " | "))
   607→        var box := PanelContainer.new()
   608→        var sbf := StyleBoxFlat.new()
   609→        sbf.bg_color = Color(0.25, 0.05, 0.05, 0.94)
   610→        sbf.set_corner_radius_all(12)
   611→        box.add_theme_stylebox_override("panel", sbf)
   612→        box.set_anchors_preset(Control.PRESET_FULL_RECT)
   613→        box.mouse_filter = Control.MOUSE_FILTER_IGNORE
   614→        var lbl := _mk_label(msg, 22, true, false, Color(1, 0.9, 0.85))
   615→        lbl.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
   616→        box.add_child(lbl)
   617→        add_child(box)

   619→func _show(p: PanelContainer) -> void:
   620→        if panel_home == null or panel_levels == null:
   621→                return
   622→        panel_home.visible = p == panel_home
   623→        panel_levels.visible = p == panel_levels
   624→        # کانتینر مأموریت‌ها هم وقتی پنل مخفی است باید مخفی شود
   625→        # (دو لایه محافظت: mouse_filter=IGNORE + مخفی‌سازی کامل)
   626→        # ماشین‌های تزئینی و دکمه‌های تپ فقط مال صفحه‌ی خانه‌اند؛
   627→        # وگرنه روی پنل مأموریت‌ها می‌افتند و تپِ دکمه‌ها را می‌دزدند
   628→        for n in deco_cars:
   629→                if is_instance_valid(n):
   630→                        n.visible = p == panel_home
   631→        if levels_center != null:
   632→                levels_center.visible = p == panel_levels
   633→        _refresh()
