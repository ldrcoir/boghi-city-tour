   608→func _show(p: PanelContainer) -> void:
   609→        if panel_home == null or panel_levels == null:
   610→                return
   611→        panel_home.visible = p == panel_home
   612→        panel_levels.visible = p == panel_levels
   613→        # تیتر فقط مال صفحه‌ی خانه است (لیست ماموریت‌ها خودش هدر دارد)
   614→        for n in title_nodes:
   615→                if is_instance_valid(n):
   616→                        n.visible = p == panel_home
   617→        for n in deco_cars:
   618→                if is_instance_valid(n):
   619→                        n.visible = p == panel_home
   620→        # هر باز شدن مأموریت‌ها: تیکه تازه‌ی اکبر
   621→        if p == panel_levels and akbar_line_label != null:
   622→                var line := CarBrain.pick_line("akbar", "idle")
   623→                if line.is_empty():
   624→                        line = "امروز جاده مال ماست!"
   625→                akbar_line_label.text = line
   626→                if brain != null and akbar_rect != null and is_instance_valid(akbar_rect) and akbar_rect.texture != null:
   627→                        brain.say(akbar_rect, "akbar", "idle")
   628→        _refresh()
