   524→   973→## گیج سرعت — قوس + عقربه، حس داشبورد
   525→   974→class SpeedGauge extends Control:
   526→   975→        var game: Node2D
   527→   976→        var frac := 0.0
   528→   977→
   529→   978→        func _process(_delta: float) -> void:
   530→   979→                if game == null:
   531→   980→                        return
   532→   981→                var tgt: float = clampf(absf(game.vel.length()) / (game.max_s * 1.45), 0.0, 1.0)
   533→   982→                frac = lerpf(frac, tgt, _delta * 8.0)
   534→   983→                queue_redraw()

   536→   985→        func _draw() -> void:
   537→   986→                var c := size * 0.5
   538→   987→                c.y = size.y * 0.86
   539→   988→                var r := minf(size.x * 0.46, size.y * 0.8)
   540→   989→                var a0 := PI * 0.78
   541→   990→                var a1 := PI * 2.22
   542→   991→                draw_arc(c, r, a0, a1, 40, Color(0.1, 0.1, 0.16, 0.75), 10.0, true)
   543→   992→                var ac := Color(0.25, 0.75, 1.0).lerp(Color(1.0, 0.35, 0.15), frac)
   544→   993→                if frac > 0.02:
   545→   994→                        draw_arc(c, r, a0, a0 + (a1 - a0) * frac, 40, ac, 10.0, true)
   546→   995→                var na := a0 + (a1 - a0) * frac
   547→   996→                var tip := c + Vector2(cos(na), sin(na)) * (r - 18.0)
