   746→func _make_shadow_tex() -> ImageTexture:
   747→        var sz := Vector2i(150, 90)
   748→        var img := Image.create(sz.x, sz.y, false, Image.FORMAT_RGBA8)
   749→        var cx := sz.x / 2.0
   750→        var cy := sz.y / 2.0
   751→        for y in sz.y:
   752→                for x in sz.x:
   753→                        var d := Vector2((x - cx) / (cx - 5.0), (y - cy) / (cy - 5.0)).length()
   754→                        var a: float = clamp(1.0 - d, 0.0, 1.0)
   755→                        img.set_pixel(x, y, Color(0.0, 0.0, 0.02, a * a * 0.85))
   756→        return ImageTexture.create_from_image(img)
