   483→## صدای لاله‌زبان — یک جمله‌ی بی‌کلامِ آهنگین که شخصیتِ ماشین را می‌سازد
   484→func _speak(text: String, car_id: String, mood: String) -> void:
   485→        if not voice_enabled:
   486→                return # پیش‌فرض سکوت — ریتم طنز از متن می‌آید نه از بوق سینوسی
   487→        var v: Dictionary = PERSONAS[car_id]["voice"] if PERSONAS.has(car_id) else {"pitch": 300.0, "tone": 0.5, "rate": 10.0, "jit": 0.15}
   488→        var pitch := float(v["pitch"])
   489→        var tone := float(v["tone"])
   490→        var rate := float(v["rate"])
   491→        var jit := float(v["jit"])
   492→        var syl_dur := 1.0 / rate
   493→        var dur := clampf(0.25 + text.length() * 0.045, 0.45, 2.4)
   494→        var syls := int(ceil(dur / syl_dur))
   495→        var sr := 22050
   496→        var n := int(dur * sr)
   497→        var data := PackedByteArray()
   498→        data.resize(n * 2)
   499→        var phase := 0.0
   500→        for k in syls:
   501→                var syl_pitch := pitch * (1.0 + _rng.randf_range(-jit, jit))
   502→                if mood == "happy":
   503→                        syl_pitch *= 1.0 + 0.06 * float(k) / float(syls) # آهنگین رو به بالا
   504→                elif mood == "grumpy":
   505→                        syl_pitch *= 1.0 - 0.05 * float(k) / float(syls)
   506→                var step := 2.0 * PI * syl_pitch / float(sr)
   507→                var s0 := int(float(k) * syl_dur * sr)
   508→                var s1: int = min(n, int(float(k + 1) * syl_dur * sr) - int(sr * 0.02))
   509→                for i in range(s0, s1):
   510→                        var tt := float(i - s0) / float(sr)
   511→                        var in_syl := float(i - s0) / float(syl_dur * sr)
   512→                        var env := pow(maxf(0.0, sin(PI * clampf(in_syl, 0.0, 1.0))), 0.65)
   513→                        if tt < 0.01 or tt > syl_dur - 0.03:
   514→                                env = 0.0
   515→                        var s := sin(phase)
   516→                        var sq := signf(s)
   517→                        var wob := 1.0 + 0.02 * sin(2.0 * PI * 6.0 * tt)
   518→                        var amp := env * 0.34 * wob
   519→                        var val := (1.0 - tone) * s + tone * sq
   520→                        var iv := int(clampf(val * amp, -1.0, 1.0) * 32767.0)
   521→                        data.encode_s16(i * 2, iv)
   522→                        phase += step
   523→        var wav := AudioStreamWAV.new()
   524→        wav.format = AudioStreamWAV.FORMAT_16_BITS
   525→        wav.mix_rate = sr
   526→        wav.stereo = false
   527→        wav.data = data
   528→        voice_player.stream = wav
   529→        voice_player.pitch_scale = 1.0
   530→        voice_player.play()
